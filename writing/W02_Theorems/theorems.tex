\begin{abstract}
\noindent This note states the contraction theorems for the BQTF posterior as Theory proved them on 23 November, and the one theorem still in progress. It replaces the $\sigma$ discussion of the W-01 note: the theorems below cover $\sigma$ \emph{learned} under the paper's IG prior, with no condition $\sigma\ge\sbar$. \stPA{} (proof complete; checked by an auditor from the other model family);

\section{The model and the method}
\label{sec:model}

\paragraph{Data and truth.} $y_1,\dots,y_n$ are observed at $u_i=i/n$. At a fixed level $\tau\in(0,1)$ the truth $\theta^*\in\R^n$ is the vector of conditional $\tau$-quantiles. Write $Z_i:=y_i-\theta^*_i$, $F_i$ for the cdf of $Z_i$ (so $F_i(0)=\tau$), $\rho_\tau(x)=x(\tau-\ind\{x<0\})$ for the check loss, $Q(\theta):=\sum_i\rho_\tau(y_i-\theta_i)$, $t:=\tau(1-\tau)$, $D:=\Dthree$ (third differences, $m=n-3$ rows), $\TV(\theta):=n^2\norm{D\theta}_1$, $\delta:=\theta-\theta^*$, $\norm{\delta}_n^2:=n^{-1}\sum_i\delta_i^2$ and $\Delta_n^2(\delta):=n^{-1}\sum_i\min(\delta_i^2,|\delta_i|)$.

\paragraph{Felipe's method (the paper's default).}
\[
\pi(\theta,\lambda,\sigma\mid y)\ \propto\ \sigma^{-n}e^{-Q(\theta)/\sigma}\cdot\lambda^{m}e^{-\lambda\norm{D\theta}_1}\cdot\Ga(\lambda;1,b_0)\cdot\IG(\sigma;0.1,0.1),
\]
with a flat prior on quadratics (the null space of $D$).
\begin{itemize}
\item \emph{Likelihood:} the asymmetric Laplace (ALD) working likelihood with scale $\sigma$.
\item \emph{Trend prior:} Laplace on $D\theta$ with rate $\lambda$ (the paper's $k=2$).
\item \emph{$\lambda$:} learned, $\lambda\sim\Ga(1,b_0)$ with a constant rate $b_0=10^{-3}$ (VS) or $5\times10^{-4}$ (S).
\item \emph{$\sigma$:} learned, $\sigma\sim\IG(0.1,0.1)$.
\item \emph{Joint version:} several levels ($\tau\in\{0.90,0.93,0.96,0.97,0.98,0.99\}$ in the paper) fitted together, with the posterior restricted to the non-crossing set $C:=\{\Theta:\theta_{i1}\le\dots\le\theta_{iq}\text{ for all }i\}$.
\end{itemize}
All theorems concern this \emph{posterior}. They say nothing about how well a particular sampler reaches it.

\paragraph{The variants the theorems cover.}
\begin{itemize}
\item \textbf{V0T} (``his method plus a floor''): Felipe's method with $\lambda$ restricted to $\lambda\ge\Lambda_-:=\kappa_-n^{5/2}$, $\kappa_-=1.6\times10^{-5}$ (the Gamma prior truncated below). At $n=500$ the floor is $89$; the smallest posterior $5\%$ quantile of $\lambda$ over 90 fits was $267$ (\ME, memo). So at the paper's $n$ the floor never binds.
\item \textbf{A-rule}: Felipe's method with one change, $\lambda\sim\Ga(1,b_n)$, $b_n=1/(5500\,(n/500)^{5/2})$. It depends on $n$ only, is the same for both signals, and has no truncation.
\item \textbf{Oracle $\lambda$} (Theorems C, C$^*$): $\sigma$ learned as in the method, $\lambda$ a deterministic sequence $\lambda\asymp n^{5/2}/V$.
\end{itemize}

\section{Assumptions}
\label{sec:assumptions}

All constants are fixed; ``$\sup_P$'' is over every law of $y$ that satisfies the listed assumptions with these constants.
\begin{description}[leftmargin=0pt]
\item[(A1)] $y_1,\dots,y_n$ are independent.
\item[(A2)] Two-sided secant margin at the quantile: $f|x|\le|F_i(x)-F_i(0)|\le\bar f|x|$ for $|x|\le L$, all $i$. ``Upper (A2)'' is the right-hand inequality alone.
\item[(A3)] $\TV(\theta^*)=n^2\norm{D\theta^*}_1\le V$.
\item[(UI)] $\rho_\tau(Z_i)$ is uniformly integrable on row average, with a common modulus.
\item[(S0)] $d\le\szero\le B$ with $d>0$, where $\szero:=n^{-1}\sum_i\E\rho_\tau(Z_i)$ is the population check risk (the value the learned $\sigma$ converges to).
\end{description}
No exponential moments, no (EM) and no condition $\sigma\ge\sbar=t/f$ appear in Section~\ref{sec:main}.

\noindent\textit{For the paper's designs} (Gaussian, Beta, mixed normal; VS and S truths), (A1)--(A3), (UI) and (S0) hold at every fixed level; $\szero\in[0.0046,0.29]$ (W-01 note, Tables 4--6; THEOREM\_C.md). The learned $\sigma$ sits at $0.13$--$0.64\,\sbar$ (\ME, memo), so results that need $\sigma\ge\sbar$ (Theorem B of W-01) do not reach it.

\section{Main results: proved and cross-family audited}
\label{sec:main}

\subsection{\texorpdfstring{Theorem C: $\sigma$ learned, $\lambda$ deterministic, rate $n^{-4/7}$}{Theorem C: sigma learned, lambda deterministic}}

\plainly{With $\sigma$ learned exactly as in the paper and $\lambda$ set by a deterministic rule near the oracle, the posterior of the trend contracts, in sup-norm too, and the learned $\sigma$ settles on the population check risk $\szero$. The rate $n^{-4/7}$ is slower than the oracle rate; Theorem C$^*$ improves it.}

\begin{named}{Theorem C}
Model: $\pi(\theta,\sigma\mid y)\propto\sigma^{-n}e^{-Q(\theta)/\sigma}\,e^{-\lambda\norm{D\theta}_1}\,\IG(\sigma;a,b)$, flat prior on quadratics, any fixed $a,b>0$, penalty free of $\sigma$. Assume (A1), (A2) (both halves, constants $L,f,\bar f$), (A3), (UI) and $\szero\ge d>0$. Let $\lambda$ be deterministic, $\lambda\asymp n^{5/2}/V$; more generally any deterministic $\lambda$ with $n^{23/11}(\log n)^{12/11}\ll\lambda\ll n^{108/41}V^{-42/41}$ for which C1's denominator hypothesis (ii) holds. With $r_n^2:=C\,n^{1/7}\lambda^{-2/7}(\log n)^{6/7}$, i.e.\ $r_n^2\asymp n^{-4/7}(\log n)^{6/7}$ at the oracle, for every fixed $\eta>0$:
\begin{enumerate}[label=\arabic*.]
\item $\supP\Pi(\norm{\theta-\theta^*}_n^2>r_n^2\mid y)\to0$;
\item $\supP\Pi(\norm{\theta-\theta^*}_\infty>\eta\mid y)\to0$;
\item $\supP\Pi(|\sigma/\szero-1|>\eta\mid y)\to0$.
\end{enumerate}
\end{named}
\covers{$\sigma$ exactly as the method learns it. $\lambda$ is not learned.}

Final read by Fable: ACCEPT at the oracle; the window's upper end was corrected to $n^{108/41}V^{-42/41}$. File \texttt{THEOREM\_C.md}, commit \texttt{574407c}.}

\subsection{\texorpdfstring{Theorem C$^*$: $\sigma$ learned, oracle $\lambda$, rate $n^{-5/6}$}{Theorem C*: sigma learned, oracle rate}}

\plainly{The same setting as Theorem C, now at the oracle rate $n^{-5/6}$ (up to $(\log n)^{1/6}$). This is the first theorem at the best rate that learns $\sigma$ the way the paper does. It needs no light tails and no lower bound on $\sigma$.}

\begin{named}{Theorem C$^*$}
Model and assumptions of Theorem C: fixed interior $\tau$; (A1); (A2) in secant form with $(L,f,\bar f)$; (A3) with $V$; (UI) with a common modulus; $d\le\szero\le B$, $d>0$. $\lambda$ deterministic, $a_1n^{5/2}/V\le\lambda\le a_2n^{5/2}/V$. Put $\varepsilon^2:=C(\log n)^{1/6}V^{1/3}n^{-5/6}$. Then for every $M>M^*$ (an explicit constant, \texttt{FABLE\_A\_n56.md} \S2.1) and every fixed $\eta>0$:
\begin{enumerate}[label=\arabic*.]
\item $\supP\Pi(\norm{\theta-\theta^*}_n^2>M\varepsilon^2\mid y)\to0$, i.e.\ rate $n^{-5/6}(\log n)^{1/6}$;
\item $\supP\Pi(\norm{\theta-\theta^*}_\infty>s^*\mid y)\to0$, with $s^*:=6\sqrt M\varepsilon+7.06(\sqrt M\varepsilon)^{4/5}T_0^{1/5}$, $T_0:=Tn^3/\lambda$, $T\ge4$; $s^*\asymp n^{-7/30}(\log n)^{1/15}$;
\item $\supP\Pi(|\sigma/\szero-1|>\eta\mid y)\to0$.
\end{enumerate}
No (EM), no $\sigma\ge\sbar$, no moments beyond uniform integrability.
\end{named}
\covers{$\sigma$ exactly as the method learns it, at the oracle rate; $\lambda$ deterministic.}
\status{\stPA. Assembly by Theory, written independently by Fable A (\texttt{FABLE\_A\_n56.md}). Inputs Lemma F, W$'$ and C2 by Fable, audited by Opus (ACCEPT; C2 shells ACCEPT WITH FIXES). Cross-family audit by Astra: Theorem C$^*$ ACCEPT. Files \texttt{THEOREM\_C\_STAR.md}, \texttt{FABLE\_A\_n56.md}; commit \texttt{4350530}.}

\subsection{\texorpdfstring{Theorems V0T and V0T$^*$: his method plus a $\lambda$ floor}{Theorems V0T and V0T*: his method plus a lambda floor}}
\label{sec:v0t}

\plainly{Felipe's method, with $\sigma$ \emph{and} $\lambda$ learned, plus one change: $\lambda$ may not fall below $1.6\times10^{-5}n^{5/2}$. Then the trend contracts at the oracle rate $n^{-5/6}(\log n)^{1/6}$, and the learned $\sigma$ settles on $\szero$. At $n=500$ the floor never binds, so there the method is unchanged. This is the main theorem available now.}

\noindent\textbf{Model (both theorems).} $\pi(\theta,\lambda,\sigma\mid y)\propto\sigma^{-n}e^{-Q(\theta)/\sigma}\,\lambda^me^{-\lambda\norm{D\theta}_1}\,p_n(\lambda)\,\IG(\sigma;a,b)$, flat prior on quadratics, $a,b>0$ fixed, $p_n$ a probability law on $[\Lambda_-,\infty)$, $\Lambda_-:=\kappa_-n^{5/2}$, $\kappa_->0$ fixed. $J_0:=[\Lambda_-,2\Lambda_-]$, $p_{0,n}:=p_n(J_0)$.
\begin{itemize}
\item V0T (Felipe + floor): $p_n=\Ga(1,b_0)$ truncated to $[\Lambda_-,\infty)$, $b_0$ constant; $p_{0,n}=1-e^{-b_0\Lambda_-}\to1$.
\item Scaled version: $b_n=c_bn^{-5/2}$; $p_{0,n}=1-e^{-c_b\kappa_-}$, a constant.
\item In general: any $p_n$ on $[\Lambda_-,\infty)$ with $-\log p_{0,n}=o(n\bar\varepsilon^2)$.
\end{itemize}
\textbf{Assumptions (both).} Those of Theorem C$^*$, with no condition on $\lambda$ and no relation between $\kappa_-$ and $V$.

\begin{named}{Theorem V0T}
Fix $T\ge4$; $T_0:=Tn^3/\Lambda_-=(T/\kappa_-)\sqrt n$; $r^2:=C_r(T/\kappa_-)^{2/7}n^{-4/7}(\log n)^{6/7}$. For every fixed $\eta>0$:
\begin{enumerate}[label=\arabic*.]
\item $\supP\Pi(\norm{\theta-\theta^*}_n>r\text{ or }\TV(\theta-\theta^*)>T_0\mid y)\to0$;
\item $\supP\Pi(\norm{\theta-\theta^*}_\infty>s_K\mid y)\to0$, $s_K:=6r+7.06\,r^{4/5}T_0^{1/5}\asymp n^{-9/70}(\log n)^{12/35}$;
\item $\supP\Pi(|\sigma/\szero-1|>\eta\mid y)\to0$.
\end{enumerate}
No (EM), no $\sigma\ge\sbar$, no moments beyond uniform integrability, and no containment statement for $\lambda$ is needed or claimed.
\end{named}

\begin{named}{Theorem V0T$^*$}
Put $\bar\varepsilon^2:=C_\varepsilon n^{-5/6}[\kappa_-^{-1/3}(\log m)^{1/6}+\kappa_-^{5/3}V^2]$. For every $M>\bar M^*$ (an explicit constant, \texttt{FABLE\_A\_n56.md} \S3.1) and every fixed $\eta>0$:
\begin{enumerate}[label=\arabic*.]
\item $\supP\Pi(\norm{\theta-\theta^*}_n^2>M\bar\varepsilon^2\mid y)\to0$, i.e.\ rate $n^{-5/6}(\log n)^{1/6}$, with constant $\propto\kappa_-^{-1/3}+\kappa_-^{5/3}V^2/(\log n)^{1/6}$;
\item $\supP\Pi(\norm{\theta-\theta^*}_\infty>\bar s^*\mid y)\to0$, $\bar s^*:=6\sqrt M\bar\varepsilon+7.06(\sqrt M\bar\varepsilon)^{4/5}T_0^{1/5}\asymp n^{-7/30}(\log n)^{1/15}$;
\item $\supP\Pi(|\sigma/\szero-1|>\eta\mid y)\to0$.
\end{enumerate}
No containment statement for $\lambda$ is needed or claimed. At $\kappa_-=1/V$ the constant is the oracle's $V^{1/3}$; the theorem is rate-optimal for every $\kappa_-$, but adaptive to $V$ only in that case.
\end{named}
\covers{V0T: Felipe's method (constant $b_0$) plus the floor $\kappa_-=1.6\times10^{-5}$; also the scaled schedule. Rate $n^{-4/7}(\log n)^{6/7}$ (V0T) and $n^{-5/6}(\log n)^{1/6}$ (V0T$^*$).}

V0T$^*$: \stPA. Prover Fable A. Auditor Astra: ACCEPT WITH FIXES (eight editorial findings, all held; Lemma C2-mix ACCEPT WITH FIXES). File \texttt{FABLE\_A\_n56.md} \S3, commit \texttt{afae478}.}

\subsection{\texorpdfstring{Theorem J: the joint non-crossing posterior}{Theorem J: the joint non-crossing posterior}}

\plainly{For a fixed set of levels, the paper's joint non-crossing posterior inherits V0T$^*$ (or C$^*$) at every level at once, at $n^{-5/6}(\log n)^{1/6}$, with no extra separation condition. The reason is that the true quantile curves are separated (upper (A2) forces it) and each level's posterior concentrates in sup-norm, so the ordering constraint becomes inactive. So the theorem says the constraint does no harm asymptotically. It does not say the constraint helps, and at the paper's $n$ the constraint is probably still active.}

\begin{named}{Theorem J}
Assumptions: $q\ge2$ fixed levels; for each $j$ the assumptions of Theorem V0T$^*$ hold at level $\tau_j$ ((A1); full (A2) in secant form; (A3); (UI) of $\rho_{\tau_j}(y_i-\theta^*_{ij})$; $d_j\le\sigma_{0,n,j}\le B_j$; floor $\Lambda_{-,j}=\kappa_{-,j}n^{5/2}$ and $p_n$ as in V0T$^*$, constant or scaled $b_0$), with common constants. $\Pi^\otimes$ is the product of the single-level posteriors; $\Pi_C$ is the joint posterior on $C$; $\bar\varepsilon_j^2$, $\bar M^*_j$, $\bar s^*_j$ are those of V0T$^*$ at level $j$; $M>\max_j\bar M^*_j$. For every fixed $\eta>0$, as $n\to\infty$:
\begin{enumerate}[label=\arabic*.]
\item $\supP\Pi^\otimes(C^c\mid y)\to0$; equivalently $\supP\norm{\Pi_C-\Pi^\otimes}_{TV}\to0$;
\item $\supP\Pi_C(\norm{\theta_j-\theta^*_j}_n^2>M\bar\varepsilon_j^2\text{ for some }j\mid y)\to0$: rate $n^{-5/6}(\log n)^{1/6}$ at every level simultaneously;
\item $\supP\Pi_C(\norm{\theta_j-\theta^*_j}_\infty>\bar s^*_j\text{ for some }j\mid y)\to0$;
\item $\supP\Pi_C(|\sigma_j/\sigma_{0,n,j}-1|>\eta\text{ for some }j\mid y)\to0$.
\end{enumerate}
The same holds with Theorem C$^*$ in place of V0T$^*$. Separation is implied, not assumed (Lemma J0): $\theta^*_{i,j+1}-\theta^*_{ij}\ge g_0:=\min(L,\Delta\tau/\bar f)$.
\end{named}
\covers{The paper's joint non-crossing fit, with V0T$^*$ (or C$^*$) at each level.}
\status{\stPA. Prover Fable B. Auditor Astra: ACCEPT WITH FIXES. File \texttt{FABLE\_B\_joint.md}, commit \texttt{6697527}. Scope: fixed levels only. Levels that approach each other with $n$, and the realistic-$n$ regime where the constraint is active, are SKETCH/CONJECTURE. The argument needs $n\gtrsim(c\bar f/\Delta\tau)^{30/7}$, far beyond the paper's $n$ for level gaps of a few hundredths.}

\subsection{\texorpdfstring{Theorem LB: the matching lower bound}{Theorem LB: the matching lower bound}}

\plainly{The posterior cannot contract faster than $\lambda^{-1/3}$. At the oracle $\lambda\asymp n^{5/2}/V$ this is $n^{-5/6}$, so together with C$^*$ or V0T$^*$ the exponent $5/6$ is exact. The only remaining gap is the factor $(\log n)^{1/6}$ on the upper side. The lower bound needs very little: independence and the upper half of (A2).}

\begin{named}{Theorem LB}
Assume (A1) and upper (A2) (constants $L,\bar f$; $C_1=\max(\bar f/2,1/L)$). Let $n\ge5$, $\omega=1/\sigma>0$, $\lambda>0$, and let $(\theta^*,\lambda)$ be $\beta$-admissible (a shift budget $\beta$, defined in \texttt{FABLE\_B\_lowerbound.md} \S7). Put
\[
\Sigma:=100+\beta+\omega(1.44C_1+1)+c_\omega,\qquad c_0:=(932\,\Sigma)^{-5},\qquad D:=n\lambda^{-1/3}.
\]
If $(32\Sigma)^3\le\lambda\le(n/42)^3$, then
\[
\E_0\,\Pi\big(\Delta_n^2(\theta-\theta^*)\le c_0\lambda^{-1/3}\mid y\big)\ \le\ 1.44/D+8D^{3/2}\exp\{5-(\Sigma+1)D\}.
\]
Along any sequence with $\lambda_n\to\infty$, $\lambda_n=o(n^3)$ and $\beta$ bounded, the left side tends to $0$, uniformly over the laws with the given $(L,\bar f)$ and the $\beta$-admissible truths. The same holds for $\norm{\theta-\theta^*}_n^2$. No lower margin, no (EM), no moments, no $\sigma\ge\sbar$.
\end{named}

\noindent\textbf{Exact scope.}
\begin{itemize}
\item \emph{Noise:} (A1) and upper (A2) only.
\item \emph{$\sigma$:} any fixed $\sigma>0$ ($c_0$ depends on $\sigma$); uniformly on compact $\sigma$-intervals; and $\sigma$ learned under IG, given Lemma S's hypotheses ((UI), $\szero\ge d$, (A3) or a bounded truth) plus a bounded shift budget, $\lambda\to\infty$ and $D\to\infty$.
\item \emph{$\lambda$, deterministic:} under (A3), every $\lambda_n\to\infty$ with $\lambda_n\le a_2n^{5/2}/V$; for quadratic truths, every $\lambda_n=o(n^3)$. Above the oracle both tuning restrictions must be re-checked at the growing budget, and they can fail (e.g.\ $\lambda=n^{59/20}$ is not certified).
\item \emph{$\lambda$, learned:} his method plus the floor (V0T: constant $b_0$, fixed $V$ and $\kappa_-$) gives $\Pi(\Delta_n^2\le c\Lambda_-^{-1/3}\mid y)\to0$ with a constant $c$, log-free. Not covered: unfloored scaled hyperpriors (A-rule, V2).
\item \emph{Centre:} a ball around the truth $\theta^*$, not around a posterior-dependent centre.
\end{itemize}

\noindent\textbf{Two-sided statement at the oracle.} For $a_1n^{5/2}/V\le\lambda\le a_2n^{5/2}/V$, with posterior mass $\to1$,
\[
c_0(V/a_2)^{1/3}n^{-5/6}\ \le\ \Delta_n^2(\theta-\theta^*)\ \le\ M(\log n)^{1/6}V^{1/3}n^{-5/6},
\]
in $\Delta_n^2$ and in $\norm{\cdot}_n^2$. The upper half needs more than LB: full (A2), (A3), (UI) and $\szero\ge d$, through C$^*$ (learned $\sigma$, oracle $\lambda$), V0T$^*$ (his method + floor), the fixed-scale steps of C$^*$ (a fixed $\sigma$), or Theorem B (a fixed $\sigma$ in the (EM) range). Under (A1) and upper (A2) alone, only the lower half is established.
\covers{Any fixed or learned $\sigma$; deterministic $\lambda$; V0T. Not the A-rule.}

\subsection{\texorpdfstring{The no-floor obstruction}{The no-floor obstruction}}
\label{sec:obstruction}

\plainly{Without a floor (or another change), no theorem can cover Felipe's method for \emph{every} law in the class. There is one bounded, smooth noise law at $\tau=0.05$, inside the class of C, C$^*$ and V0T$^*$, on which the exact no-floor posterior fails to contract: fits that nearly interpolate the data carry more evidence than the truth. The floor is \emph{one} repair that works. It is not shown to be the only one. The paper's own designs are not of this kind: they meet the necessary condition with a wide margin.}

\begin{named}{Obstruction (necessary condition)}
Write $k=\min(\tau,1-\tau)$, $R_n=\norm{Dy}_1/n$. Suppose that with probability tending to one $R_n\le r$ and $Q(\theta^*)/n\ge s_*$, for fixed $r>0$ and $s_*>2tr$. For every $2tr<q<s_*$ there is a fixed $c>0$ such that, on these data events and for all large $n$, $\Pi(Q(\theta)\ge nq\mid y)\le e^{-cn}$. Consequently, when $R_n\to R>0$ and $Q(\theta^*)/n\to\sigma_0$ in probability, consistency of the exact posterior requires
\[
\sigma_0\ \le\ 2\tau(1-\tau)R .
\]
\end{named}
\noindent\textbf{Counterexample.} $\tau=0.05$, $\theta^*=0$, i.i.d.\ errors from the uniform mixture $0.049\cdot U[-0.02,-0.01]+0.002\cdot U[-0.005,0.005]+0.949\cdot U[1,1.001]$. It satisfies (A1), full (A2) ($L=0.004$, slopes $0.2$), (A3) and (UI), with $\sigma_0/\sbar\approx0.20$. For every fixed $b_0>0$ and the IG$(0.1,0.1)$ prior, the exact joint posterior puts exponentially vanishing mass on a fixed neighbourhood of the truth. The same holds for the unfloored scaled rate $b_n=c\,n^{-5/2}$.

\noindent\textbf{The paper's designs.} The ratio $\sigma_0/(2\tau(1-\tau)R)$ is $0.22$--$0.38$ (Gaussian), $0.18$--$0.47$ (Beta) and $0.22$--$0.38$ (mixed), at $\tau\in\{0.01,0.05,0.25,0.5,0.75,0.95,0.99\}$ (computed: MC for $R$, quadrature for $\sigma_0$; \texttt{noFloor\_necessary.out}, commit \texttt{51b7163}). So the obstruction does not apply to them. \textbf{Whether the condition, with margin, is also sufficient is \OP.}
\covers{Shows that the unmodified method (constant $b_0$, or unfloored $b_n\propto n^{-5/2}$) has no class-wide theorem.}
\status{\stPA. Prover Astra (an assistant). Auditor Fable: ACCEPT WITH FIXES (19 findings, all held; the wording ``the floor excludes exactly these configurations'' was withdrawn). File \texttt{astra\_noFloor.md}, commit \texttt{7baa5b4}. The Gaussian margin is confirmed in closed form; the Beta and mixed margins are numerical.}

\subsection{\texorpdfstring{Order $k$: the rate $n^{-(2k+1)/(2k+2)}$}{Order k}}

\plainly{The same chain works for every trend-filtering order $k\ge1$. The oracle tuning is $\lambda=n^{k+1/2}/V$ and the rate is $n^{-(2k+1)/(2k+2)}$ ($k=2$: $n^{-5/6}$). At $k=0$ (the fused lasso) only Theorem B survives.}

\noindent For $k\ge1$, with $D=D^{(k+1)}$, a flat prior on polynomials of degree $\le k$ and (A3$_k$): $n^k\norm{D\theta^*}_1\le V$, the oracle tuning is $\lambda_{k,n}=n^{k+1/2}/V$ and
\[
\varepsilon_n^2=C(\log n)^{1/(2k+2)}V^{1/(k+1)}n^{-(2k+1)/(2k+2)}.
\]
C1, C2 and V0T/V2T with the floor $\Lambda_-=\kappa_-n^{k+1/2}$ carry over for $k\ge1$ by substitution into the audited proofs; Lemma S carries over for every $k\ge0$. At $k=1$ the C1 window margin is thin ($n^{1/14}(\log n)^{-4/7}$) but sufficient. At $k=0$ sup-norm localization fails on the proposed body; only Theorem B (under (EM)) survives.

\section{Target: the A-rule without a floor \quad \stTARGET}
\label{sec:arule}

\plainly{The A-rule changes one hyperparameter of Felipe's method and adds no floor. Its rate $b_n$ shrinks with $n$ so that the prior keeps pace with the oracle $\lambda$. Eight links lead from the proved pieces to the target. Five are proved, the assembly is proved, and three are open (L2, L4a, L4b). \textbf{It is not a theorem until the open links close.}}

\begin{named}{Target (A-rule, no floor) --- not a theorem}
Model: Felipe's method with $\lambda\sim\Ga(1,b_n)$, $b_n=1/(5500\,(n/500)^{5/2})$, $\sigma\sim\IG(0.1,0.1)$ learned, no truncation. At $\tau\in[0.05,0.95]$, for the paper's three noise designs, $\theta$ contracts at $n^{-5/6}$ up to logs (via L7: $n^{-5/6}(\log n)^{1/3}$), with $\sigma$ and $\lambda$ both learned.
\end{named}

\noindent\textbf{Why it must be design-specific.} By the obstruction (Section~\ref{sec:obstruction}) and the soft-floor notes (Astra; Fable B, unaudited), no class-wide theorem exists for an unfloored adaptive Gamma prior. So the target is for the paper's three designs only, at $\tau\in[0.05,0.95]$.

\noindent\textbf{Route.} Corollary P (Theorem H file) gives, for every event $B$ and every data vector,

\Pi(B\mid y)\ \le\ \Pi(B\mid y,\lambda\ge\Lambda_-)+\Pi(\lambda<\Lambda_-\mid y).

The first term is V0T$^*$ with an $n$-dependent floor (L7, proved). Everything else is the lower-tail input
\[
\textbf{(LT)}\qquad \supP\Pi(\lambda<\Lambda_{-,n}\mid y)\to0,\qquad \Lambda_{-,n}=c_-n^{5/2}(\log n)^{-1/2}.
\]

\begin{named}{Theorem H (Huber-body localization above any floor)}
At fixed $\sigma$ ($\omega=1/\sigma$), let $p$ be any probability law on $[\Lambda_-,\infty)$, $J_0=[\Lambda_-,2\Lambda_-]$, $p_0:=p(J_0)>0$. Assume (A1), full (A2) in secant form, (A3) with $V$; no moments, no (EM), no condition on $\sigma$. With the Huber function $h_L$, $\mathbf h(\delta)=\sum_ih_L(\delta_i)$, $T\ge4$, $T_0:=Tn^3/\Lambda_-$ and $r^2:=C_HK^{12/7}(T/\Lambda_-)^{2/7}(\log2n)^{6/7}$, $C_H:=(605/f)^{12/7}$: for $K=K_n\to\infty$ with $K_n^6(\log n)^3=o(\Lambda_-)$, $\Lambda_-\le n^2$ and $-\log p_0=o(nr^2)$,
\[
\supP\sup_{\omega\in[\omega_1,\omega_2]}\Pi_\omega\big(\mathbf h(\theta-\theta^*)/n>r^2\text{ or }\TV(\theta-\theta^*)>T_0\mid y\big)\to0 .
\]

\section{What is not covered}
\label{sec:notcovered}

\begin{itemize}
\item \textbf{Felipe's exact method (no floor, constant $b_0$).} There is no contraction theorem at any rate. Class-wide, none can exist (the obstruction). Under constant $b_0$ the learned $\lambda$ grows like $n^{3/4}$ (\ME, memo), outside every window. A design-specific theorem for it is \OP.
\item \textbf{Any result without a floor, class-wide.} Impossible for unfloored Gamma priors, constant or scaled (the obstruction). Soft floors through Gamma shape parameters either pin $\lambda$ at its prior mean or fail class-wide (Astra; Fable B; not audited).
\item \textbf{$\tau=0.01$ and $0.99$ under the A-rule.} $\bar\beta\approx13$--$14$ against $6$ for all three designs, so the sufficient condition of this route is not met there; nothing is claimed at these levels for the A-rule. The $\bar\beta$ condition does not enter V0T$^*$ or Theorem J.
\item \textbf{The joint version of the A-rule.} Theorem J is proved for V0T$^*$ and C$^*$ only.
\item \textbf{The joint fit at the paper's $n$.} Theorem J works by showing that the constraint becomes inactive. Where the constraint is active (realistic $n$, nearby levels), nothing is proved.
\item \textbf{Coverage of credible bands; Bernstein--von Mises.} Not claimed anywhere. $\sbar$ matters for calibration, not for the rate (memo, item (a)).
\item \textbf{Lower bound around the posterior's own centre}, and a sharp lower bound above the oracle: new lemmas needed. The upper side keeps a $(\log n)^{1/6}$ factor.
\item \textbf{A posterior-mean (point estimate) corollary.} Kept as a low-tier item; it needs posterior uniform integrability.
\item \textbf{Order $k=0$} (fused lasso): only Theorem B, under (EM).
\item \textbf{Size of $n_0$.} The sample size beyond which the bounds are in force is not quantified (low tier). The sup-norm condition (N10) of V0T$^*$ is the slowest: $n\gtrsim s_0^{-70/9}$ up to logs.
\item \textbf{The no-floor sufficiency question.} Whether $\sigma_0\le2\tau(1-\tau)R$ with margin suffices on the paper's designs is \OP.
\end{itemize}
